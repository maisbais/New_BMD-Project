  id = ~SDMVPSU,

  weights = ~WTMEC6YR,  
  nest = TRUE,
  data = eval_missing_data)

  data = design_missing_eval,
  by = analytic_status,
  include = c(RIDAGEYR, RIAGENDR, RIDRETH3, fpl_cat, parent_edu, FSDCH, BMXHT, screen_quart),
  statistic = list(
    all_continuous()  ~ "{mean} ({mean.std.error})",
    all_categorical() ~ "{n_unweighted} ({p}%)"
  ),
  missing = "no"
) %>%
  add_p() %>%
  modify_header(stat_1 ~ "**Analytic Sample (Included)**<br>n = {n_unweighted}") %>%
